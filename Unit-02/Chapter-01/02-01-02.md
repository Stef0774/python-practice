wind_speed = int(input())
water_temperature = float(input())
lightning_seen = input()

if wind_speed >= 60 or lightning_seen == "yes":
    print("beach closed")
elif wind_speed >= 35 or water_temperature < 8:
    print("no swimming")
elif wind_speed >= 20 and water_temperature <= 16:
    print("swim with caution")
else:
    print("all clear")
